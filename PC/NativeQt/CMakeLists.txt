cmake_minimum_required(VERSION 3.21)
project(MotorTestPlatformNative VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Widgets SerialPort PrintSupport Test)
add_library(qcustomplot STATIC third_party/qcustomplot/qcustomplot.cpp third_party/qcustomplot/qcustomplot.h)
target_include_directories(qcustomplot PUBLIC third_party/qcustomplot)
target_link_libraries(qcustomplot PUBLIC Qt6::Widgets Qt6::PrintSupport)
add_library(platform_core STATIC src/wire_codec.cpp src/history.cpp src/serial_endpoint.cpp src/monitor_link.cpp src/operator_codec.cpp src/operator_client.cpp src/plan_run.cpp src/plan_controller.cpp src/strict_json.cpp src/native_audit.cpp src/serial_text.cpp)
target_include_directories(platform_core PUBLIC src)
target_link_libraries(platform_core PUBLIC Qt6::Core Qt6::SerialPort)
add_executable(MotorTestPlatformNative WIN32 src/main.cpp src/main_window.cpp src/trend_plot.cpp src/plan_panel.cpp src/serial_panel.cpp)
target_link_libraries(MotorTestPlatformNative PRIVATE platform_core qcustomplot)
target_compile_definitions(MotorTestPlatformNative PRIVATE PLATFORM_VERSION="${PROJECT_VERSION}")
file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/profiles")
configure_file("${CMAKE_CURRENT_SOURCE_DIR}/../Devices/profiles/c620_m3508_lab_v1.json" "${CMAKE_BINARY_DIR}/profiles/c620_m3508_lab_v1.json" COPYONLY)
file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/docs")
configure_file("${CMAKE_CURRENT_SOURCE_DIR}/USER_GUIDE.md" "${CMAKE_BINARY_DIR}/docs/USER_GUIDE.md" COPYONLY)
if(MSVC)
  target_compile_options(platform_core PRIVATE /W4 /utf-8)
  target_compile_options(MotorTestPlatformNative PRIVATE /W4 /utf-8)
else()
  target_compile_options(platform_core PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(MotorTestPlatformNative PRIVATE -Wall -Wextra -Wpedantic)
endif()
enable_testing()
add_executable(native_protocol_tests tests/protocol_tests.cpp)
target_link_libraries(native_protocol_tests PRIVATE platform_core Qt6::Test)
target_compile_definitions(native_protocol_tests PRIVATE
  NATIVE_VECTOR_FILE="${CMAKE_CURRENT_SOURCE_DIR}/tests/synthetic-wire-vectors.json"
  NATIVE_HISTORY_FILE="${CMAKE_CURRENT_SOURCE_DIR}/tests/synthetic-feedback.jsonl")
add_test(NAME native_protocol COMMAND native_protocol_tests)
add_executable(native_monitor_tests tests/monitor_tests.cpp)
target_link_libraries(native_monitor_tests PRIVATE platform_core Qt6::Test)
target_compile_definitions(native_monitor_tests PRIVATE NATIVE_VECTOR_FILE="${CMAKE_CURRENT_SOURCE_DIR}/tests/synthetic-wire-vectors.json")
add_test(NAME native_monitor COMMAND native_monitor_tests)
add_executable(native_operator_tests tests/operator_tests.cpp)
target_link_libraries(native_operator_tests PRIVATE platform_core Qt6::Test)
target_compile_definitions(native_operator_tests PRIVATE NATIVE_OPERATOR_VECTORS="${CMAKE_CURRENT_SOURCE_DIR}/tests/operator-vectors.json")
add_test(NAME native_operator COMMAND native_operator_tests)
add_executable(native_plan_tests tests/plan_tests.cpp)
target_link_libraries(native_plan_tests PRIVATE platform_core Qt6::Test)
target_compile_definitions(native_plan_tests PRIVATE NATIVE_PROFILE="${CMAKE_CURRENT_SOURCE_DIR}/../Devices/profiles/c620_m3508_lab_v1.json" NATIVE_PLAN_SYNTHETIC_VECTORS="${CMAKE_CURRENT_SOURCE_DIR}/tests/synthetic-wire-vectors.json")
add_test(NAME native_plan COMMAND native_plan_tests)
add_executable(native_controller_tests tests/controller_tests.cpp)
target_link_libraries(native_controller_tests PRIVATE platform_core Qt6::Test)
target_compile_definitions(native_controller_tests PRIVATE NATIVE_PROFILE="${CMAKE_CURRENT_SOURCE_DIR}/../Devices/profiles/c620_m3508_lab_v1.json")
add_test(NAME native_controller COMMAND native_controller_tests)
add_executable(native_audit_tests tests/audit_tests.cpp)
target_link_libraries(native_audit_tests PRIVATE platform_core Qt6::Test)
target_compile_definitions(native_audit_tests PRIVATE NATIVE_PROFILE="${CMAKE_CURRENT_SOURCE_DIR}/../Devices/profiles/c620_m3508_lab_v1.json")
add_test(NAME native_audit COMMAND native_audit_tests)
add_executable(native_serial_tool_tests tests/serial_tool_tests.cpp src/serial_panel.cpp src/main_window.cpp src/plan_panel.cpp src/trend_plot.cpp)
target_link_libraries(native_serial_tool_tests PRIVATE platform_core qcustomplot Qt6::Widgets Qt6::Test)
target_compile_definitions(native_serial_tool_tests PRIVATE NATIVE_VECTOR_FILE="${CMAKE_CURRENT_SOURCE_DIR}/tests/synthetic-wire-vectors.json")
target_compile_definitions(native_serial_tool_tests PRIVATE NATIVE_PROFILE="${CMAKE_CURRENT_SOURCE_DIR}/../Devices/profiles/c620_m3508_lab_v1.json" NATIVE_RELEASE_PLANS="${CMAKE_CURRENT_SOURCE_DIR}/plans")
add_test(NAME native_serial_tool COMMAND native_serial_tool_tests)
